#2. Write a function that computes the player point rating (1 point for wins, 0.5 points for draws, 0 points for defeats). How are point ratings distributed? Why?

point_rating <- function(d) {
  bind_rows(data.frame(id = d$White, points = d$Score),
            data.frame(id = d$Black, points = 1 - d$Score)) |>
    group_by(id) |>
    summarise(points = sum(points), games = n())
}

players <- point_rating(df)
summary(players$points)
##    Min. 1st Qu.  Median    Mean 3rd Qu.    Max. 
##    0.00    1.00    3.00    8.91    9.00  167.50
ggplot(players, aes(x = points)) +
  geom_histogram(bins = 50) +
  labs(x = "Punti totali", y = "Numero di giocatori")

I punti hanno una distribuzione asimmetrica a destra, la maggior parte dei giocatori ha pochi punti, mentre pochi giocatori ne hanno moltissimi. Succede perché i punti dipendono dal numero di partite giocate, e poche persone giocano tante partite.

#3. An interesting property of Elo’s ratings is that the sum of all player ratings is always 0. Formally show this property. (Hint: use the fact that \(s_{i,j} + s_{j,i}=1\) and \(\mu_{i,j} + \mu_{j,i} =1\))

zeta <- 400
kappa <- 25
mu <- function(ri, rj) 1 / (1 + 10^(-(ri - rj) / zeta))

g <- expand.grid(s_ij = c(0, 0.5, 1), ri = seq(-1000, 1000, 10), rj = seq(-1000, 1000, 10))
g$s_ji <- 1 - g$s_ij                      

delta_S <- kappa * (g$s_ij - mu(g$ri, g$rj)) + kappa * (g$s_ji - mu(g$rj, g$ri))
max(abs(delta_S))

#4. Write a function that computes the player Elo rating (set \(\zeta = 400\) and \(\kappa = 25\)) (Hint: take advantage of the 0-sum property to optimize the code). Obserse its distribution: why is different from the point distribution? Verify the 0-sum property for Elo’s ratings

elo_rating <- function(d, zeta = 400, kappa = 25) {
  id <- unique(c(d$White, d$Black))
  w <- match(d$White, id)
  b <- match(d$Black, id)
  r <- numeric(length(id))
  for (k in seq_len(nrow(d))) {
    delta <- kappa * (d$Score[k] - 1 / (1 + 10^((r[b[k]] - r[w[k]]) / zeta)))
    r[w[k]] <- r[w[k]] + delta   # il Bianco guadagna delta
    r[b[k]] <- r[b[k]] - delta   # il Nero perde delta (0-sum)
  }
  data.frame(id, elo = r)
}
